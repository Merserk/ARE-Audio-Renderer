cmake_minimum_required(VERSION 3.28)
project(ASIORenderEngine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_subdirectory(third_party/soxr-build)

add_library(are_core STATIC src/core/format.cpp src/core/signal_processor.cpp src/core/resampler.cpp)
target_include_directories(are_core PUBLIC src)
target_include_directories(are_core SYSTEM PRIVATE third_party/r8brain)
target_link_libraries(are_core PRIVATE are_soxr)
if(MSVC)
  target_compile_options(are_core PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
else()
  target_compile_options(are_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(WIN32)
  enable_language(RC)
  add_library(are_platform STATIC
    src/win/devices.cpp src/win/settings.cpp src/win/clock.cpp
    src/win/engine.cpp src/win/media.cpp)
  target_include_directories(are_platform PUBLIC src third_party/asio/common)
  target_link_libraries(are_platform PUBLIC are_core ole32 oleaut32 strmiids
    uuid advapi32 user32)
  target_compile_definitions(are_platform PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN
    NOMINMAX _WIN32_WINNT=0x0A00)

  add_library(ASIORenderEngine SHARED
    src/win/filter.cpp src/win/property_page.cpp src/win/registration.cpp
    src/win/dll.cpp src/win/exports.def resources/renderer.rc)
  target_link_libraries(ASIORenderEngine PRIVATE are_platform comctl32)
  set_target_properties(ASIORenderEngine PROPERTIES PREFIX "" OUTPUT_NAME "ARE-Audio-Renderer")

  add_executable(arectl src/tools/arectl.cpp resources/control.rc)
  set_target_properties(arectl PROPERTIES OUTPUT_NAME "ARE-Audio-Renderer-Control")
  target_link_libraries(arectl PRIVATE are_platform)
  add_executable(ASIORenderSettings WIN32 src/tools/settings_main.cpp resources/settings.rc)
  set_target_properties(ASIORenderSettings PROPERTIES OUTPUT_NAME "ARE-Audio-Renderer-Settings")
  target_link_libraries(ASIORenderSettings PRIVATE are_platform comctl32)
  foreach(target are_platform ASIORenderEngine arectl ASIORenderSettings)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /guard:cf)
    target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  endforeach()
  foreach(target arectl ASIORenderSettings)
    target_link_options(${target} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest")
  endforeach()
  install(TARGETS ASIORenderEngine arectl ASIORenderSettings RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION lib EXCLUDE_FROM_ALL)
  install(FILES third_party/ASIO-GPL-3.0.txt DESTINATION . RENAME LICENSE)
  install(FILES LICENSE DESTINATION licenses RENAME ARE-MIT.txt)
  install(FILES THIRD_PARTY_NOTICES.md DESTINATION .)
  install(FILES resources/Release-README.md DESTINATION . RENAME README.md)
  install(FILES scripts/Install.ps1 scripts/Install.cmd scripts/Uninstall.ps1 scripts/Uninstall.bat DESTINATION .)
  install(FILES third_party/asio/LICENSE.txt DESTINATION licenses RENAME ASIO-SDK.txt)
  install(FILES third_party/r8brain/LICENSE DESTINATION licenses RENAME r8brain-MIT.txt)
  install(FILES third_party/r8brain/OOURA-FFT-NOTICE.txt DESTINATION licenses)
  install(FILES third_party/soxr/COPYING.LGPL DESTINATION licenses RENAME SoX-LGPL-2.1.txt)
  install(FILES third_party/soxr/LICENCE third_party/soxr-build/FFT-NOTICES.txt DESTINATION licenses)
endif()

if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE are_core)
  add_test(NAME lossless_conversion_and_queue COMMAND core_tests)
  add_executable(signal_tests tests/signal_tests.cpp)
  target_link_libraries(signal_tests PRIVATE are_core)
  add_test(NAME volume_and_transition_signals COMMAND signal_tests)
  add_executable(playback_rate_tests tests/playback_rate_tests.cpp)
  target_link_libraries(playback_rate_tests PRIVATE are_core)
  add_test(NAME playback_speed_and_pitch COMMAND playback_rate_tests)
  add_executable(resampler_tests tests/resampler_tests.cpp)
  target_link_libraries(resampler_tests PRIVATE are_core)
  add_test(NAME sample_rate_conversion_and_pcm_precision COMMAND resampler_tests)
  add_test(NAME sample_rate_conversion_scalar COMMAND resampler_tests)
  set_tests_properties(sample_rate_conversion_scalar PROPERTIES ENVIRONMENT "SOXR_USE_SIMD64=0")
  if(WIN32)
    add_library(fake_asio_driver SHARED tests/fake_asio.cpp tests/fake_driver_dll.cpp tests/fake_driver.def)
    target_link_libraries(fake_asio_driver PRIVATE are_platform)
    add_executable(engine_tests tests/engine_tests.cpp tests/fake_asio.cpp)
    target_link_libraries(engine_tests PRIVATE are_platform)
    add_test(NAME asio_engine_contract COMMAND engine_tests)
    add_executable(filter_tests tests/filter_tests.cpp)
    target_link_libraries(filter_tests PRIVATE are_platform)
    add_dependencies(filter_tests fake_asio_driver ASIORenderEngine)
    add_test(NAME directshow_filter_contract COMMAND filter_tests $<TARGET_FILE:ASIORenderEngine> $<TARGET_FILE:fake_asio_driver>)
    set_tests_properties(asio_engine_contract PROPERTIES TIMEOUT 30)
    set_tests_properties(directshow_filter_contract PROPERTIES TIMEOUT 60)
  endif()
endif()
